package com.example.gestureplayer

import android.Manifest
import android.content.Intent
import android.content.pm.PackageManager
import android.graphics.BitmapFactory
import android.media.AudioManager
import android.net.Uri
import android.os.Bundle
import android.widget.*
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import androidx.camera.view.PreviewView
import androidx.core.content.ContextCompat
import com.bumptech.glide.Glide
import com.google.android.material.dialog.MaterialAlertDialogBuilder
import java.util.concurrent.TimeUnit

class MainActivity : AppCompatActivity() {

    private lateinit var previewView: PreviewView
    private lateinit var albumArt: ImageView
    private lateinit var songTitle: TextView
    private lateinit var artistName: TextView
    private lateinit var currentTime: TextView
    private lateinit var totalTime: TextView
    private lateinit var seekBar: SeekBar
    private lateinit var playPauseBtn: ImageButton
    private lateinit var previousBtn: ImageButton
    private lateinit var nextBtn: ImageButton
    private lateinit var cameraToggleBtn: ImageButton
    private lateinit var libraryBtn: ImageButton
    private lateinit var backgroundBtn: ImageButton
    private lateinit var mainLayout: RelativeLayout

    private lateinit var audioManager: AudioManager
    private var musicPlayer: MusicPlayer? = null
    private var gestureController: GestureController? = null
    private var isCameraActive = false

    private val backgroundColors = intArrayOf(
        ContextCompat.getColor(this, R.color.background),
        ContextCompat.getColor(this, R.color.secondary),
        ContextCompat.getColor(this, R.color.secondary_dark)
    )

    private val permissionLauncher = registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { permissions ->
        if (permissions.values.all { it }) {
            setupMusicPlayer()
        } else {
            Toast.makeText(this, "دسترسی داده نشد", Toast.LENGTH_LONG).show()
        }
    }

    private val backgroundPicker = registerForActivityResult(
        ActivityResultContracts.GetContent()
    ) { uri: Uri? ->
        uri?.let {
            mainLayout.setBackground(
                BitmapFactory.decodeStream(contentResolver.openInputStream(it))
            )
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        initializeViews()
        audioManager = getSystemService(AUDIO_SERVICE) as AudioManager
        requestPermissions()
    }

    private fun initializeViews() {
        previewView = findViewById(R.id.previewView)
        albumArt = findViewById(R.id.albumArt)
        songTitle = findViewById(R.id.songTitle)
        artistName = findViewById(R.id.artistName)
        currentTime = findViewById(R.id.currentTime)
        totalTime = findViewById(R.id.totalTime)
        seekBar = findViewById(R.id.seekBar)
        playPauseBtn = findViewById(R.id.playPauseBtn)
        previousBtn = findViewById(R.id.previousBtn)
        nextBtn = findViewById(R.id.nextBtn)
        cameraToggleBtn = findViewById(R.id.cameraToggleBtn)
        libraryBtn = findViewById(R.id.libraryBtn)
        backgroundBtn = findViewById(R.id.backgroundBtn)
        mainLayout = findViewById(R.id.mainLayout)

        playPauseBtn.setOnClickListener {
            musicPlayer?.togglePlayPause()
            updatePlayPauseIcon()
        }
        previousBtn.setOnClickListener { musicPlayer?.playPrevious() }
        nextBtn.setOnClickListener { musicPlayer?.playNext() }

        cameraToggleBtn.setOnClickListener { toggleCamera() }
        libraryBtn.setOnClickListener { startActivity(Intent(this, LibraryActivity::class.java)) }
        backgroundBtn.setOnClickListener { showBackgroundChooser() }

        seekBar.setOnSeekBarChangeListener(object : SeekBar.OnSeekBarChangeListener {
            override fun onProgressChanged(seekBar: SeekBar?, progress: Int, fromUser: Boolean) {
                if (fromUser) musicPlayer?.seekTo(progress.toLong() * 1000)
            }
            override fun onStartTrackingTouch(seekBar: SeekBar?) {}
            override fun onStopTrackingTouch(seekBar: SeekBar?) {}
        })
    }

    private fun requestPermissions() {
        val permissions = mutableListOf(
            Manifest.permission.READ_MEDIA_AUDIO,
            Manifest.permission.CAMERA,
            Manifest.permission.POST_NOTIFICATIONS
        )
        val needed = permissions.filter {
            ContextCompat.checkSelfPermission(this, it) != PackageManager.PERMISSION_GRANTED
        }
        if (needed.isNotEmpty()) {
            permissionLauncher.launch(needed.toTypedArray())
        } else {
            setupMusicPlayer()
            toggleCamera()
        }
    }

    private fun setupMusicPlayer() {
        musicPlayer = MusicPlayer(this)
        musicPlayer?.let { player ->
            if (player.getAllSongs().isNotEmpty()) {
                updateUI(player.getCurrentSong())
                player.playIndex(0)

                Thread {
                    while (!isFinishing) {
                        runOnUiThread {
                            val song = player.getCurrentSong()
                            val position = player.getCurrentPosition()
                            seekBar.max = song?.let { (it.duration / 1000).toInt() } ?: 100
                            seekBar.progress = position
                            currentTime.text = formatTime(position.toLong())
                            totalTime.text = song?.let { formatTime(it.duration) } ?: "00:00"
                        }
                        Thread.sleep(500)
                    }
                }.start()
            } else {
                Toast.makeText(this, "هیچ آهنگی پیدا نشد", Toast.LENGTH_LONG).show()
            }
        }
    }

    private fun toggleCamera() {
        isCameraActive = !isCameraActive
        if (isCameraActive) {
            gestureController = GestureController(this, previewView, audioManager) { gesture ->
                handleGesture(gesture)
            }
            gestureController?.start()
            cameraToggleBtn.setImageResource(R.drawable.ic_camera_active)
            Toast.makeText(this, "کنترل حرکتی فعال شد", Toast.LENGTH_SHORT).show()
        } else {
            gestureController?.stop()
            gestureController = null
            cameraToggleBtn.setImageResource(R.drawable.ic_camera)
            Toast.makeText(this, "کنترل حرکتی غیرفعال شد", Toast.LENGTH_SHORT).show()
        }
    }

    private fun handleGesture(gesture: GestureType) {
        when (gesture) {
            GestureType.NEXT -> {
                musicPlayer?.playNext()
                Toast.makeText(this, "آهنگ بعدی", Toast.LENGTH_SHORT).show()
            }
            GestureType.PREVIOUS -> {
                musicPlayer?.playPrevious()
                Toast.makeText(this, "آهنگ قبلی", Toast.LENGTH_SHORT).show()
            }
            GestureType.VOLUME_UP -> adjustVolume(true)
            GestureType.VOLUME_DOWN -> adjustVolume(false)
            else -> {}
        }
    }

    private fun adjustVolume(increase: Boolean) {
        val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC)
        val currentVol = audioManager.getStreamVolume(AudioManager.STREAM_MUSIC)
        val newVol = if (increase) {
            (currentVol + 1).coerceAtMost(maxVol)
        } else {
            (currentVol - 1).coerceAtLeast(0)
        }
        audioManager.setStreamVolume(AudioManager.STREAM_MUSIC, newVol, 0)
    }

    private fun updateUI(song: Song?) {
        song?.let {
            songTitle.text = it.title
            artistName.text = it.artist
            totalTime.text = formatTime(it.duration)
            it.albumArtUri?.let { artUri ->
                try {
                    Glide.with(this)
                        .load(artUri)
                        .placeholder(R.drawable.ic_music)
                        .error(R.drawable.ic_music)
                        .into(albumArt)
                } catch (e: Exception) {
                    albumArt.setImageResource(R.drawable.ic_music)
                }
            }
        }
    }

    private fun updatePlayPauseIcon() {
        if (musicPlayer?.isPlaying() == true) {
            playPauseBtn.setImageResource(R.drawable.ic_pause)
        } else {
            playPauseBtn.setImageResource(R.drawable.ic_play)
        }
    }

    private fun showBackgroundChooser() {
        MaterialAlertDialogBuilder(this)
            .setTitle("تغییر پس‌زمینه")
            .setItems(arrayOf("پس‌زمینه تیره", "پس‌زمینه مشکی", "انتخاب از گالری")) { _, which ->
                when (which) {
                    0 -> mainLayout.setBackgroundColor(backgroundColors[0])
                    1 -> mainLayout.setBackgroundColor(backgroundColors[2])
                    2 -> backgroundPicker.launch("image/*")
                }
            }
            .show()
    }

    private fun formatTime(millis: Long): String {
        return String.format("%02d:%02d",
            TimeUnit.MILLISECONDS.toMinutes(millis),
            TimeUnit.MILLISECONDS.toSeconds(millis) -
                TimeUnit.MINUTES.toSeconds(TimeUnit.MILLISECONDS.toMinutes(millis))
        )
    }

    override fun onDestroy() {
        super.onDestroy()
        gestureController?.stop()
        musicPlayer?.release()
    }
}

package com.example.gestureplayer

import android.content.Context
import android.media.MediaPlayer
import android.net.Uri
import android.provider.MediaStore

data class Song(
    val id: Long,
    val title: String,
    val artist: String,
    val album: String,
    val duration: Long,
    val uri: Uri,
    val albumArtUri: Uri?
)

class MusicPlayer(private val context: Context) {
    private var mediaPlayer: MediaPlayer? = null
    private var songList: MutableList<Song> = mutableListOf()
    private var currentIndex = 0

    init {
        loadSongsFromDevice()
    }

    private fun loadSongsFromDevice() {
        val projection = arrayOf(
            MediaStore.Audio.Media._ID,
            MediaStore.Audio.Media.TITLE,
            MediaStore.Audio.Media.ARTIST,
            MediaStore.Audio.Media.ALBUM,
            MediaStore.Audio.Media.DURATION,
            MediaStore.Audio.Media.ALBUM_ID
        )

        val selection = "${MediaStore.Audio.Media.IS_MUSIC} != 0"
        val sortOrder = "${MediaStore.Audio.Media.TITLE} ASC"

        try {
            context.contentResolver.query(
                MediaStore.Audio.Media.EXTERNAL_CONTENT_URI,
                projection,
                selection,
                null,
                sortOrder
            )?.use { cursor ->
                val idColumn = cursor.getColumnIndexOrThrow(MediaStore.Audio.Media._ID)
                val titleColumn = cursor.getColumnIndexOrThrow(MediaStore.Audio.Media.TITLE)
                val artistColumn = cursor.getColumnIndexOrThrow(MediaStore.Audio.Media.ARTIST)
                val albumColumn = cursor.getColumnIndexOrThrow(MediaStore.Audio.Media.ALBUM)
                val durationColumn = cursor.getColumnIndexOrThrow(MediaStore.Audio.Media.DURATION)
                val albumIdColumn = cursor.getColumnIndexOrThrow(MediaStore.Audio.Media.ALBUM_ID)

                while (cursor.moveToNext()) {
                    val id = cursor.getLong(idColumn)
                    val albumId = cursor.getLong(albumIdColumn)
                    val contentUri = MediaStore.Audio.Media.EXTERNAL_CONTENT_URI
                    val uri = Uri.withAppendedPath(contentUri, id.toString())
                    val albumArtUri = Uri.parse("content://media/external/audio/albumart/$albumId")

                    songList.add(
                        Song(
                            id = id,
                            title = cursor.getString(titleColumn) ?: "نامشخص",
                            artist = cursor.getString(artistColumn) ?: "نامشخص",
                            album = cursor.getString(albumColumn) ?: "نامشخص",
                            duration = cursor.getLong(durationColumn),
                            uri = uri,
                            albumArtUri = albumArtUri
                        )
                    )
                }
            }
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    fun playIndex(index: Int) {
        if (songList.isEmpty() || index !in songList.indices) return
        currentIndex = index
        playCurrentSong()
    }

    fun playNext() {
        if (songList.isEmpty()) return
        currentIndex = (currentIndex + 1) % songList.size
        playCurrentSong()
    }

    fun playPrevious() {
        if (songList.isEmpty()) return
        currentIndex = if (currentIndex > 0) currentIndex - 1 else songList.size - 1
        playCurrentSong()
    }

    private fun playCurrentSong() {
        val song = songList[currentIndex]
        try {
            mediaPlayer?.release()
            mediaPlayer = MediaPlayer.create(context, song.uri)
            mediaPlayer?.setOnCompletionListener { playNext() }
            mediaPlayer?.start()
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    fun togglePlayPause() {
        mediaPlayer?.let {
            if (it.isPlaying) it.pause() else it.start()
        }
    }

    fun seekTo(seconds: Long) {
        mediaPlayer?.seekTo((seconds * 1000).toInt())
    }

    fun getCurrentSong(): Song? = if (songList.isEmpty()) null else songList[currentIndex]
    fun getAllSongs(): List<Song> = songList
    fun isPlaying(): Boolean = mediaPlayer?.isPlaying ?: false
    fun getCurrentPosition(): Int = mediaPlayer?.currentPosition ?: 0
    fun pause() { mediaPlayer?.pause() }

    fun release() {
        mediaPlayer?.release()
        mediaPlayer = null
    }
}

package com.example.gestureplayer

import android.content.Context
import android.media.AudioManager
import android.util.Log
import androidx.camera.core.*
import androidx.camera.lifecycle.ProcessCameraProvider
import androidx.camera.view.PreviewView
import androidx.core.content.ContextCompat
import com.google.mediapipe.framework.image.MPImageBuilder
import com.google.mediapipe.tasks.core.BaseOptions
import com.google.mediapipe.tasks.vision.core.RunningMode
import com.google.mediapipe.tasks.vision.handlandmarker.HandLandmarker
import com.google.mediapipe.tasks.vision.handlandmarker.HandLandmarkerResult
import java.util.concurrent.Executors

enum class GestureType {
    NEXT, PREVIOUS, VOLUME_UP, VOLUME_DOWN, NONE
}

class GestureController(
    private val context: Context,
    private val previewView: PreviewView,
    private val audioManager: AudioManager,
    private val onGesture: (GestureType) -> Unit
) {
    private var handLandmarker: HandLandmarker? = null
    private val executor = Executors.newSingleThreadExecutor()

    private var lastMovementDirection = 0
    private var lastSwipeTime = 0L
    private val indexFingerPoints = ArrayDeque<Pair<Float, Float>>()

    fun start() {
        setupHandLandmarker()
        startCamera()
    }

    private fun setupHandLandmarker() {
        try {
            val modelBuffer = loadModelFile(context)
            val options = HandLandmarker.HandLandmarkerOptions.builder()
                .setBaseOptions(
                    BaseOptions.builder()
                        .setModelAssetBuffer(modelBuffer)
                        .build()
                )
                .setRunningMode(RunningMode.LIVE_STREAM)
                .setNumHands(1)
                .setMinHandDetectionConfidence(0.7f)
                .setMinHandPresenceConfidence(0.7f)
                .setResultListener { result: HandLandmarkerResult, _: Long ->
                    processLandmarks(result)
                }
                .build()

            handLandmarker = HandLandmarker.createFromOptions(context, options)
        } catch (e: Exception) {
            Log.e("GestureController", "Error: ${e.message}")
        }
    }

    private fun loadModelFile(context: Context): java.nio.ByteBuffer {
        val inputStream = context.assets.open("hand_landmarker.task")
        val bytes = inputStream.readBytes()
        return java.nio.ByteBuffer.wrap(bytes)
    }

    private fun startCamera() {
        val cameraProviderFuture = ProcessCameraProvider.getInstance(context)
        cameraProviderFuture.addListener({
            try {
                val cameraProvider = cameraProviderFuture.get()
                val preview = Preview.Builder().build()
                preview.setSurfaceProvider(previewView.surfaceProvider)

                val imageAnalysis = ImageAnalysis.Builder()
                    .setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
                    .setOutputImageFormat(ImageAnalysis.OUTPUT_IMAGE_FORMAT_RGBA_8888)
                    .build()

                imageAnalysis.setAnalyzer(executor) { imageProxy ->
                    val mpImage = MPImageBuilder.fromMediaImage(
                        imageProxy.image!!,
                        imageProxy.imageInfo.rotationDegrees
                    ).build()
                    handLandmarker?.detectAsync(mpImage, System.currentTimeMillis())
                    imageProxy.close()
                }

                cameraProvider.unbindAll()
                cameraProvider.bindToLifecycle(
                    context as androidx.lifecycle.LifecycleOwner,
                    CameraSelector.DEFAULT_FRONT_CAMERA,
                    preview,
                    imageAnalysis
                )
            } catch (e: Exception) {
                Log.e("GestureController", "Error: ${e.message}")
            }
        }, ContextCompat.getMainExecutor(context))
    }

    private fun processLandmarks(result: HandLandmarkerResult) {
        if (result.landmarks().isNotEmpty()) {
            val landmarks = result.landmarks()[0]
            if (landmarks.size >= 21) {
                val wrist = landmarks[0]
                val indexTip = landmarks[8]

                detectHorizontalSwipe(wrist.x(), wrist.y())
                detectCircularMotion(indexTip.x(), indexTip.y())
            }
        }
    }

    private fun detectHorizontalSwipe(wristX: Float, wristY: Float) {
        val customWristX = (wristX - 0.5f) * 2

        if (customWristX > 0.4f) {
            val currentTime = System.currentTimeMillis()
            if (lastMovementDirection != 1 && currentTime - lastSwipeTime > 1000) {
                onGesture(GestureType.NEXT)
                lastMovementDirection = 1
                lastSwipeTime = currentTime
            }
        } else if (customWristX < -0.4f) {
            val currentTime = System.currentTimeMillis()
            if (lastMovementDirection != -1 && currentTime - lastSwipeTime > 1000) {
                onGesture(GestureType.PREVIOUS)
                lastMovementDirection = -1
                lastSwipeTime = currentTime
            }
        } else {
            lastMovementDirection = 0
        }
    }

    private fun detectCircularMotion(indexX: Float, indexY: Float) {
        indexFingerPoints.addLast(Pair(indexX, indexY))
        if (indexFingerPoints.size > 60) {
            indexFingerPoints.removeFirst()
        }
        if (indexFingerPoints.size < 30) return

        val cx = indexFingerPoints.sumOf { it.first.toDouble() } / indexFingerPoints.size
        val cy = indexFingerPoints.sumOf { it.second.toDouble() } / indexFingerPoints.size

        val angles = indexFingerPoints.map { (x, y) ->
            Math.toDegrees(Math.atan2((y - cy), (x - cx))).toFloat()
        }

        var totalRotation = 0f
        for (i in 1 until angles.size) {
            var diff = angles[i] - angles[i - 1]
            if (diff > 180) diff -= 360
            if (diff < -180) diff += 360
            totalRotation += diff
        }

        if (Math.abs(totalRotation) > 300) {
            if (totalRotation > 0) {
                onGesture(GestureType.VOLUME_UP)
            } else {
                onGesture(GestureType.VOLUME_DOWN)
            }
            indexFingerPoints.clear()
        }
    }

    fun stop() {
        try {
            handLandmarker?.close()
        } catch (_: Exception) {}
        executor.shutdown()
    }
}

package com.example.gestureplayer

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.ImageView
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.bumptech.glide.Glide

class LibraryActivity : AppCompatActivity() {

    private lateinit var recyclerView: RecyclerView
    private lateinit var songList: List<Song>

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_library)

        recyclerView = findViewById(R.id.songRecyclerView)
        recyclerView.layoutManager = LinearLayoutManager(this)

        songList = MusicPlayer(this).getAllSongs()
        recyclerView.adapter = SongAdapter(songList)
    }
}

class SongAdapter(private val songs: List<Song>) :
    RecyclerView.Adapter<SongAdapter.SongViewHolder>() {

    class SongViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val albumArt: ImageView = view.findViewById(R.id.itemAlbumArt)
        val title: TextView = view.findViewById(R.id.itemTitle)
        val artist: TextView = view.findViewById(R.id.itemArtist)
        val duration: TextView = view.findViewById(R.id.itemDuration)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): SongViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_song, parent, false)
        return SongViewHolder(view)
    }

    override fun onBindViewHolder(holder: SongViewHolder, position: Int) {
        val song = songs[position]
        holder.title.text = song.title
        holder.artist.text = song.artist
        holder.duration.text = String.format("%02d:%02d",
            song.duration / 60000,
            (song.duration % 60000) / 1000
        )

        song.albumArtUri?.let { artUri ->
            try {
                Glide.with(holder.itemView.context)
                    .load(artUri)
                    .placeholder(R.drawable.ic_music)
                    .error(R.drawable.ic_music)
                    .into(holder.albumArt)
            } catch (e: Exception) {
                holder.albumArt.setImageResource(R.drawable.ic_music)
            }
        }
    }

    override fun getItemCount(): Int = songs.size
